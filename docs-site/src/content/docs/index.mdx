<Section id="speeds" index="04" eyebrow="How it works" title="One loop, three speeds" lead="The same gateway runs three loops at different speeds. The student and the teacher wear the same colours on every page.">
  <FadeContent distance={16}>
    <Flow label="The three loops">
      <FlowStep title="Every request" meta="milliseconds to seconds" tone="owned">
        The student samples its answer three times. If the samples agree and the confidence reaches the version's threshold, its answer is served; if not, the teacher answers. A few served answers are audited by the teacher.
      </FlowStep>
      <FlowStep title="Every few minutes" meta="the release gate">
        The gate reads the audits and moves a new version forward (shadow, then 10%, 50% and 100% of traffic) or rolls it back.
      </FlowStep>
      <FlowStep title="Every night" meta="Night School">
        A fresh adapter is trained on the teacher's answers only, tested on a held-out golden set, and handed to the gate.
      </FlowStep>
    </Flow>
  </FadeContent>
</Section>
