PRAGMA foreign_keys = ON;

CREATE TABLE judges (
    judge_id        INTEGER PRIMARY KEY,
    full_name       TEXT NOT NULL,
    courtroom       TEXT NOT NULL,
    appointed_date  TEXT NOT NULL,
    specialty       TEXT
);

CREATE TABLE attorneys (
    attorney_id         INTEGER PRIMARY KEY,
    full_name           TEXT NOT NULL,
    firm_name           TEXT NOT NULL,
    bar_admission_year  INTEGER NOT NULL,
    hourly_rate         REAL
);

CREATE TABLE cases (
    case_id               INTEGER PRIMARY KEY,
    case_number           TEXT NOT NULL UNIQUE,
    title                 TEXT NOT NULL,
    case_type             TEXT NOT NULL,
    filing_date           TEXT NOT NULL,
    status                TEXT NOT NULL,
    judge_id              INTEGER NOT NULL,
    plaintiff_attorney_id INTEGER NOT NULL,
    defense_attorney_id   INTEGER,
    filing_fee            REAL NOT NULL,
    claim_amount          REAL,
    closed_date           TEXT,
    FOREIGN KEY (judge_id) REFERENCES judges(judge_id),
    FOREIGN KEY (plaintiff_attorney_id) REFERENCES attorneys(attorney_id),
    FOREIGN KEY (defense_attorney_id) REFERENCES attorneys(attorney_id)
);

CREATE TABLE hearings (
    hearing_id        INTEGER PRIMARY KEY,
    case_id           INTEGER NOT NULL,
    hearing_date      TEXT NOT NULL,
    hearing_type      TEXT NOT NULL,
    outcome           TEXT,
    duration_minutes  INTEGER,
    FOREIGN KEY (case_id) REFERENCES cases(case_id)
);

### 1
DIFFICULTY: simple
QUESTION: Which civil cases are still open, and when were they filed?
SQL: SELECT case_number, title, filing_date FROM cases WHERE case_type = 'Civil' AND status = 'Open' ORDER BY filing_date;

### 2
DIFFICULTY: simple
QUESTION: What are the five biggest claims in dollar terms?
SQL: SELECT title, claim_amount FROM cases WHERE claim_amount IS NOT NULL ORDER BY claim_amount DESC LIMIT 5;

### 3
DIFFICULTY: simple
QUESTION: Which lawyers were admitted to the bar before 2008, starting with the most senior?
SQL: SELECT full_name, firm_name, bar_admission_year FROM attorneys WHERE bar_admission_year < 2008 ORDER BY bar_admission_year;

### 4
DIFFICULTY: simple
QUESTION: Which hearings ran longer than 100 minutes?
SQL: SELECT hearing_id, hearing_date, hearing_type, duration_minutes FROM hearings WHERE duration_minutes > 100 ORDER BY duration_minutes DESC;

### 5
DIFFICULTY: simple
QUESTION: Which judges have no area of specialty on record, and what courtroom are they in?
SQL: SELECT full_name, courtroom FROM judges WHERE specialty IS NULL;

### 6
DIFFICULTY: simple
QUESTION: What are the five most recently filed cases?
SQL: SELECT case_number, title, filing_date FROM cases ORDER BY filing_date DESC LIMIT 5;

### 7
DIFFICULTY: join
QUESTION: Who is the presiding judge on each criminal case that is still open?
SQL: SELECT c.case_number, c.title, j.full_name AS judge FROM cases c JOIN judges j ON j.judge_id = c.judge_id WHERE c.case_type = 'Criminal' AND c.status = 'Open' ORDER BY c.case_number;

### 8
DIFFICULTY: join
QUESTION: For which cases were arraignment hearings held, and on what dates?
SQL: SELECT h.hearing_date, c.title FROM hearings h JOIN cases c ON c.case_id = h.case_id WHERE h.hearing_type = 'Arraignment' ORDER BY h.hearing_date;

### 9
DIFFICULTY: join
QUESTION: Who represents the filing side in each family case, and which firm are they with?
SQL: SELECT c.case_number, c.title, a.full_name AS attorney, a.firm_name FROM cases c JOIN attorneys a ON a.attorney_id = c.plaintiff_attorney_id WHERE c.case_type = 'Family' ORDER BY c.case_number;

### 10
DIFFICULTY: join
QUESTION: Which cases are being defended by someone from the Public Defender Office, and by whom?
SQL: SELECT c.case_number, c.title, a.full_name AS defender FROM cases c JOIN attorneys a ON a.attorney_id = c.defense_attorney_id WHERE a.firm_name = 'Public Defender Office' ORDER BY c.case_number;

### 11
DIFFICULTY: join
QUESTION: Which cases have no defense lawyer assigned, and which judge has each one?
SQL: SELECT c.case_number, c.title, j.full_name AS judge FROM cases c JOIN judges j ON j.judge_id = c.judge_id WHERE c.defense_attorney_id IS NULL ORDER BY c.case_number;

### 12
DIFFICULTY: join
QUESTION: For hearings that ended up being continued, what was the case title, the judge, and the hearing date?
SQL: SELECT h.hearing_date, c.title, j.full_name AS judge FROM hearings h JOIN cases c ON c.case_id = h.case_id JOIN judges j ON j.judge_id = c.judge_id WHERE h.outcome = 'Continued' ORDER BY h.hearing_date;

### 13
DIFFICULTY: aggregate
QUESTION: How many cases do we have of each type?
SQL: SELECT case_type, COUNT(*) AS case_count FROM cases GROUP BY case_type ORDER BY case_count DESC;

### 14
DIFFICULTY: aggregate
QUESTION: What is the average length of a hearing for each kind of hearing, in minutes?
SQL: SELECT hearing_type, ROUND(AVG(duration_minutes), 1) AS avg_minutes FROM hearings GROUP BY hearing_type ORDER BY avg_minutes DESC;

### 15
DIFFICULTY: aggregate
QUESTION: How many cases are open versus closed, and how much in filing fees did each group bring in?
SQL: SELECT status, COUNT(*) AS case_count, SUM(filing_fee) AS total_fees FROM cases GROUP BY status;

### 16
DIFFICULTY: aggregate
QUESTION: How many cases were filed in each year?
SQL: SELECT substr(filing_date, 1, 4) AS filing_year, COUNT(*) AS case_count FROM cases GROUP BY filing_year ORDER BY filing_year;

### 17
DIFFICULTY: hard
QUESTION: Which judges are currently handling more than one open case?
SQL: SELECT j.full_name, COUNT(*) AS open_cases FROM judges j JOIN cases c ON c.judge_id = j.judge_id WHERE c.status = 'Open' GROUP BY j.judge_id, j.full_name HAVING COUNT(*) > 1;

### 18
DIFFICULTY: hard
QUESTION: Among cases that have been closed, which five took the most days from filing to closing?
SQL: SELECT case_number, title, CAST(julianday(closed_date) - julianday(filing_date) AS INTEGER) AS days_to_close FROM cases WHERE closed_date IS NOT NULL ORDER BY days_to_close DESC LIMIT 5;

### 19
DIFFICULTY: hard
QUESTION: Which cases have never had a hearing recorded?
SQL: SELECT case_number, title FROM cases WHERE case_id NOT IN (SELECT case_id FROM hearings) ORDER BY case_number;

### 20
DIFFICULTY: hard
QUESTION: For cases with a dollar claim, can you label each as large (500,000 or more), medium (50,000 to under 500,000) or small?
SQL: SELECT title, claim_amount, CASE WHEN claim_amount >= 500000 THEN 'Large' WHEN claim_amount >= 50000 THEN 'Medium' ELSE 'Small' END AS claim_size FROM cases WHERE claim_amount IS NOT NULL ORDER BY claim_amount DESC;
