PRAGMA foreign_keys = ON;

CREATE TABLE owners (
    owner_id        INTEGER PRIMARY KEY,
    first_name      TEXT NOT NULL,
    last_name       TEXT NOT NULL,
    phone           TEXT,
    email           TEXT,
    city            TEXT,
    registered_date TEXT NOT NULL
);

CREATE TABLE vets (
    vet_id      INTEGER PRIMARY KEY,
    full_name   TEXT NOT NULL,
    specialty   TEXT NOT NULL,
    hire_date   TEXT NOT NULL,
    hourly_rate REAL
);

CREATE TABLE pets (
    pet_id       INTEGER PRIMARY KEY,
    owner_id     INTEGER NOT NULL,
    name         TEXT NOT NULL,
    species      TEXT NOT NULL,
    breed        TEXT,
    birth_date   TEXT,
    weight_kg    REAL,
    microchip_id TEXT,
    FOREIGN KEY (owner_id) REFERENCES owners(owner_id)
);

CREATE TABLE visits (
    visit_id          INTEGER PRIMARY KEY,
    pet_id            INTEGER NOT NULL,
    vet_id            INTEGER NOT NULL,
    visit_date        TEXT NOT NULL,
    reason            TEXT NOT NULL,
    diagnosis         TEXT,
    fee               REAL NOT NULL,
    follow_up_needed  INTEGER NOT NULL DEFAULT 0,
    FOREIGN KEY (pet_id) REFERENCES pets(pet_id),
    FOREIGN KEY (vet_id) REFERENCES vets(vet_id)
);

### 1
DIFFICULTY: simple
QUESTION: Which dogs weigh more than 25 kg, from heaviest to lightest?
SQL: SELECT name, breed, weight_kg FROM pets WHERE species = 'Dog' AND weight_kg > 25 ORDER BY weight_kg DESC;

### 2
DIFFICULTY: simple
QUESTION: What are the five most expensive visits we've had?
SQL: SELECT visit_id, visit_date, reason, fee FROM visits ORDER BY fee DESC LIMIT 5;

### 3
DIFFICULTY: simple
QUESTION: Which pets don't have a microchip on record?
SQL: SELECT name, species FROM pets WHERE microchip_id IS NULL ORDER BY name;

### 4
DIFFICULTY: simple
QUESTION: Who are our clients in Round Rock, listed by who signed up first?
SQL: SELECT first_name, last_name, phone FROM owners WHERE city = 'Round Rock' ORDER BY registered_date ASC;

### 5
DIFFICULTY: simple
QUESTION: Which veterinarians joined in 2020 or later, newest first?
SQL: SELECT full_name, specialty, hire_date FROM vets WHERE hire_date >= '2020-01-01' ORDER BY hire_date DESC;

### 6
DIFFICULTY: simple
QUESTION: Which visits require a follow-up, in date order?
SQL: SELECT visit_id, visit_date, reason FROM visits WHERE follow_up_needed = 1 ORDER BY visit_date;

### 7
DIFFICULTY: join
QUESTION: For the visits in the second half of 2023, what was the date, the pet, the vet and the reason?
SQL: SELECT v.visit_date, p.name AS pet_name, vt.full_name AS vet_name, v.reason FROM visits v JOIN pets p ON p.pet_id = v.pet_id JOIN vets vt ON vt.vet_id = v.vet_id WHERE v.visit_date >= '2023-07-01' AND v.visit_date <= '2023-12-31' ORDER BY v.visit_date;

### 8
DIFFICULTY: join
QUESTION: Which clients own cats, and what are the cats called?
SQL: SELECT o.first_name, o.last_name, p.name AS cat_name FROM owners o JOIN pets p ON p.owner_id = o.owner_id WHERE p.species = 'Cat' ORDER BY o.last_name, p.name;

### 9
DIFFICULTY: join
QUESTION: Which vets has Bella seen, and on what dates?
SQL: SELECT vt.full_name, v.visit_date, v.reason FROM visits v JOIN pets p ON p.pet_id = v.pet_id JOIN vets vt ON vt.vet_id = v.vet_id WHERE p.name = 'Bella' ORDER BY v.visit_date;

### 10
DIFFICULTY: join
QUESTION: Are there any registered clients who haven't added a pet yet?
SQL: SELECT o.first_name, o.last_name, o.registered_date FROM owners o LEFT JOIN pets p ON p.owner_id = o.owner_id WHERE p.pet_id IS NULL;

### 11
DIFFICULTY: join
QUESTION: What did the surgeons treat, and which pets were involved?
SQL: SELECT vt.full_name, p.name AS pet_name, v.visit_date, v.diagnosis FROM visits v JOIN vets vt ON vt.vet_id = v.vet_id JOIN pets p ON p.pet_id = v.pet_id WHERE vt.specialty = 'Surgery' ORDER BY v.visit_date;

### 12
DIFFICULTY: join
QUESTION: Which pets have never been brought in for a visit, and who owns them?
SQL: SELECT p.name AS pet_name, p.species, o.first_name, o.last_name FROM pets p JOIN owners o ON o.owner_id = p.owner_id LEFT JOIN visits v ON v.pet_id = p.pet_id WHERE v.visit_id IS NULL;

### 13
DIFFICULTY: aggregate
QUESTION: How many pets do we have of each kind of animal?
SQL: SELECT species, COUNT(*) AS pet_count FROM pets GROUP BY species ORDER BY pet_count DESC, species;

### 14
DIFFICULTY: aggregate
QUESTION: How many clients do we have in each city?
SQL: SELECT city, COUNT(*) AS client_count FROM owners GROUP BY city ORDER BY client_count DESC, city;

### 15
DIFFICULTY: aggregate
QUESTION: How many visits has each vet handled and how much did those visits bring in?
SQL: SELECT vt.full_name, COUNT(v.visit_id) AS visit_count, SUM(v.fee) AS total_fees FROM vets vt JOIN visits v ON v.vet_id = vt.vet_id GROUP BY vt.vet_id, vt.full_name ORDER BY total_fees DESC;

### 16
DIFFICULTY: aggregate
QUESTION: What is the average weight for each type of animal?
SQL: SELECT species, ROUND(AVG(weight_kg), 2) AS avg_weight_kg FROM pets GROUP BY species ORDER BY avg_weight_kg DESC;

### 17
DIFFICULTY: hard
QUESTION: Which visits cost more than the average visit, and which pets were they for?
SQL: SELECT p.name AS pet_name, v.visit_date, v.reason, v.fee FROM visits v JOIN pets p ON p.pet_id = v.pet_id WHERE v.fee > (SELECT AVG(fee) FROM visits) ORDER BY v.fee DESC;

### 18
DIFFICULTY: hard
QUESTION: Which clients have had their pets in for at least two visits in total, and how much have they spent?
SQL: SELECT o.first_name, o.last_name, COUNT(v.visit_id) AS visit_count, SUM(v.fee) AS total_spent FROM owners o JOIN pets p ON p.owner_id = o.owner_id JOIN visits v ON v.pet_id = p.pet_id GROUP BY o.owner_id, o.first_name, o.last_name HAVING COUNT(v.visit_id) >= 2 ORDER BY total_spent DESC;

### 19
DIFFICULTY: hard
QUESTION: As of January 1, 2024, how many pets are young (under 3 years), adult (3 to 8 years) or senior (over 8 years)?
SQL: SELECT CASE WHEN age < 3 THEN 'Young' WHEN age <= 8 THEN 'Adult' ELSE 'Senior' END AS age_group, COUNT(*) AS pet_count FROM (SELECT (julianday('2024-01-01') - julianday(birth_date)) / 365.25 AS age FROM pets WHERE birth_date IS NOT NULL) GROUP BY age_group ORDER BY pet_count DESC;

### 20
DIFFICULTY: hard
QUESTION: For pets that have come in more than once, how many visits have they had and how many days passed between their first and latest visit?
SQL: SELECT p.name AS pet_name, COUNT(*) AS visit_count, CAST(julianday(MAX(v.visit_date)) - julianday(MIN(v.visit_date)) AS INTEGER) AS days_between_first_and_last FROM visits v JOIN pets p ON p.pet_id = v.pet_id GROUP BY p.pet_id, p.name HAVING COUNT(*) > 1 ORDER BY days_between_first_and_last DESC;
