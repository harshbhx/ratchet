PRAGMA foreign_keys = ON;

CREATE TABLE vessels (
    vessel_id      INTEGER PRIMARY KEY,
    name           TEXT NOT NULL,
    vessel_type    TEXT NOT NULL,
    home_port      TEXT NOT NULL,
    length_m       REAL,
    gross_tonnage  INTEGER,
    year_built     INTEGER,
    status         TEXT NOT NULL
);

CREATE TABLE captains (
    captain_id        INTEGER PRIMARY KEY,
    full_name         TEXT NOT NULL,
    license_number    TEXT NOT NULL UNIQUE,
    hire_date         TEXT NOT NULL,
    years_experience  INTEGER,
    phone             TEXT
);

CREATE TABLE trips (
    trip_id         INTEGER PRIMARY KEY,
    vessel_id       INTEGER NOT NULL,
    captain_id      INTEGER NOT NULL,
    departure_date  TEXT NOT NULL,
    return_date     TEXT,
    fishing_area    TEXT NOT NULL,
    fuel_liters     INTEGER,
    crew_size       INTEGER,
    FOREIGN KEY (vessel_id) REFERENCES vessels(vessel_id),
    FOREIGN KEY (captain_id) REFERENCES captains(captain_id)
);

CREATE TABLE catches (
    catch_id      INTEGER PRIMARY KEY,
    trip_id       INTEGER NOT NULL,
    species       TEXT NOT NULL,
    weight_kg     REAL NOT NULL,
    price_per_kg  REAL,
    grade         TEXT,
    FOREIGN KEY (trip_id) REFERENCES trips(trip_id)
);

### 1
DIFFICULTY: simple
QUESTION: Which boats are longer than 35 meters, from longest to shortest?
SQL: SELECT name, length_m FROM vessels WHERE length_m > 35 ORDER BY length_m DESC;

### 2
DIFFICULTY: simple
QUESTION: Who are our five most experienced captains?
SQL: SELECT full_name, years_experience FROM captains ORDER BY years_experience DESC LIMIT 5;

### 3
DIFFICULTY: simple
QUESTION: Which trips are still out at sea and haven't come back yet?
SQL: SELECT trip_id, vessel_id, departure_date, fishing_area FROM trips WHERE return_date IS NULL;

### 4
DIFFICULTY: simple
QUESTION: What are the five most recent departures?
SQL: SELECT trip_id, vessel_id, departure_date, fishing_area FROM trips ORDER BY departure_date DESC LIMIT 5;

### 5
DIFFICULTY: simple
QUESTION: Which landed catches have no quality grade recorded?
SQL: SELECT catch_id, species, weight_kg FROM catches WHERE grade IS NULL;

### 6
DIFFICULTY: simple
QUESTION: Which three catch records got the highest price per kilo?
SQL: SELECT catch_id, species, price_per_kg FROM catches ORDER BY price_per_kg DESC LIMIT 3;

### 7
DIFFICULTY: join
QUESTION: For trips that left in the first three months of 2024, show the boat name and the captain who led each one.
SQL: SELECT t.trip_id, v.name AS vessel_name, c.full_name AS captain_name, t.departure_date FROM trips t JOIN vessels v ON v.vessel_id = t.vessel_id JOIN captains c ON c.captain_id = t.captain_id WHERE t.departure_date BETWEEN '2024-01-01' AND '2024-03-31' ORDER BY t.departure_date;

### 8
DIFFICULTY: join
QUESTION: What kinds of fish have been brought in by boats based in Bergen?
SQL: SELECT DISTINCT ca.species FROM catches ca JOIN trips t ON t.trip_id = ca.trip_id JOIN vessels v ON v.vessel_id = t.vessel_id WHERE v.home_port = 'Bergen' ORDER BY ca.species;

### 9
DIFFICULTY: join
QUESTION: Which captains have taken the Santa Lucia out, and when did they leave?
SQL: SELECT c.full_name, t.departure_date FROM trips t JOIN vessels v ON v.vessel_id = t.vessel_id JOIN captains c ON c.captain_id = t.captain_id WHERE v.name = 'Santa Lucia';

### 10
DIFFICULTY: join
QUESTION: What was the market value of each catch taken in the Celtic Sea, and which boat landed it?
SQL: SELECT v.name AS vessel_name, ca.species, ROUND(ca.weight_kg * ca.price_per_kg, 2) AS catch_value FROM catches ca JOIN trips t ON t.trip_id = ca.trip_id JOIN vessels v ON v.vessel_id = t.vessel_id WHERE t.fishing_area = 'Celtic Sea' ORDER BY catch_value DESC;

### 11
DIFFICULTY: join
QUESTION: Which boats have never gone out on a trip?
SQL: SELECT v.vessel_id, v.name, v.status FROM vessels v LEFT JOIN trips t ON t.vessel_id = v.vessel_id WHERE t.trip_id IS NULL;

### 12
DIFFICULTY: join
QUESTION: For the trips currently underway, which boat is out and who is the captain?
SQL: SELECT t.trip_id, v.name AS vessel_name, c.full_name AS captain_name, t.departure_date FROM trips t JOIN vessels v ON v.vessel_id = t.vessel_id JOIN captains c ON c.captain_id = t.captain_id WHERE t.return_date IS NULL;

### 13
DIFFICULTY: aggregate
QUESTION: How many kilos of each species have we caught in total, biggest first?
SQL: SELECT species, SUM(weight_kg) AS total_kg FROM catches GROUP BY species ORDER BY total_kg DESC;

### 14
DIFFICULTY: aggregate
QUESTION: How many boats do we have of each type?
SQL: SELECT vessel_type, COUNT(*) AS vessel_count FROM vessels GROUP BY vessel_type ORDER BY vessel_count DESC;

### 15
DIFFICULTY: aggregate
QUESTION: What is the average fuel use per trip in each fishing area?
SQL: SELECT fishing_area, AVG(fuel_liters) AS avg_fuel_liters FROM trips GROUP BY fishing_area ORDER BY avg_fuel_liters DESC;

### 16
DIFFICULTY: aggregate
QUESTION: What is the average price per kilo for each quality grade, including catches that have no grade?
SQL: SELECT grade, COUNT(*) AS catch_count, ROUND(AVG(price_per_kg), 2) AS avg_price FROM catches GROUP BY grade;

### 17
DIFFICULTY: hard
QUESTION: Which species have we landed more than 20,000 kg of in total?
SQL: SELECT species, SUM(weight_kg) AS total_kg FROM catches GROUP BY species HAVING SUM(weight_kg) > 20000 ORDER BY total_kg DESC;

### 18
DIFFICULTY: hard
QUESTION: Which trips lasted more than two weeks, and how many days were they, with the boat and captain involved?
SQL: SELECT t.trip_id, v.name AS vessel_name, c.full_name AS captain_name, CAST(julianday(t.return_date) - julianday(t.departure_date) AS INTEGER) AS days_at_sea FROM trips t JOIN vessels v ON v.vessel_id = t.vessel_id JOIN captains c ON c.captain_id = t.captain_id WHERE t.return_date IS NOT NULL AND julianday(t.return_date) - julianday(t.departure_date) > 14 ORDER BY days_at_sea DESC;

### 19
DIFFICULTY: hard
QUESTION: If we call boats under 25 meters small, 25 to 40 meters medium and anything longer large, how many boats fall in each size group?
SQL: SELECT CASE WHEN length_m < 25 THEN 'small' WHEN length_m <= 40 THEN 'medium' ELSE 'large' END AS size_class, COUNT(*) AS vessel_count FROM vessels GROUP BY size_class ORDER BY vessel_count DESC;

### 20
DIFFICULTY: hard
QUESTION: Which individual catches were heavier than the average catch weight?
SQL: SELECT catch_id, species, weight_kg FROM catches WHERE weight_kg > (SELECT AVG(weight_kg) FROM catches) ORDER BY weight_kg DESC;
