PRAGMA foreign_keys = ON;

CREATE TABLE hosts (
    host_id       INTEGER PRIMARY KEY,
    full_name     TEXT NOT NULL,
    email         TEXT,
    country       TEXT,
    hourly_rate   REAL,
    joined_date   TEXT
);

CREATE TABLE shows (
    show_id       INTEGER PRIMARY KEY,
    title         TEXT NOT NULL,
    genre         TEXT,
    host_id       INTEGER NOT NULL,
    launch_date   TEXT,
    language      TEXT,
    is_active     INTEGER NOT NULL DEFAULT 1,
    FOREIGN KEY (host_id) REFERENCES hosts(host_id)
);

CREATE TABLE episodes (
    episode_id        INTEGER PRIMARY KEY,
    show_id           INTEGER NOT NULL,
    title             TEXT NOT NULL,
    episode_number    INTEGER,
    publish_date      TEXT,
    duration_minutes  INTEGER,
    downloads         INTEGER,
    guest_name        TEXT,
    FOREIGN KEY (show_id) REFERENCES shows(show_id)
);

CREATE TABLE sponsors (
    sponsor_id     INTEGER PRIMARY KEY,
    company_name   TEXT NOT NULL,
    industry       TEXT,
    contact_email  TEXT
);

CREATE TABLE ad_deals (
    deal_id            INTEGER PRIMARY KEY,
    sponsor_id         INTEGER NOT NULL,
    show_id            INTEGER NOT NULL,
    start_date         TEXT,
    end_date           TEXT,
    episodes_booked    INTEGER,
    price_per_episode  REAL,
    status             TEXT,
    FOREIGN KEY (sponsor_id) REFERENCES sponsors(sponsor_id),
    FOREIGN KEY (show_id) REFERENCES shows(show_id)
);

### 1
DIFFICULTY: simple
QUESTION: Which of our shows are in the technology category, and what language are they in?
SQL: SELECT title, language FROM shows WHERE genre = 'Technology';

### 2
DIFFICULTY: simple
QUESTION: What are the five most downloaded episodes?
SQL: SELECT title, downloads FROM episodes ORDER BY downloads DESC LIMIT 5;

### 3
DIFFICULTY: simple
QUESTION: Which of our hosts are based in the USA, and who charges the most per hour?
SQL: SELECT full_name, hourly_rate FROM hosts WHERE country = 'USA' ORDER BY hourly_rate DESC;

### 4
DIFFICULTY: simple
QUESTION: Which episodes run longer than 50 minutes and had no guest?
SQL: SELECT title, duration_minutes FROM episodes WHERE duration_minutes > 50 AND guest_name IS NULL ORDER BY duration_minutes DESC;

### 5
DIFFICULTY: simple
QUESTION: Which sponsors don't have a contact email on file?
SQL: SELECT company_name, industry FROM sponsors WHERE contact_email IS NULL ORDER BY company_name;

### 6
DIFFICULTY: simple
QUESTION: What are our three oldest shows by launch date?
SQL: SELECT title, launch_date FROM shows ORDER BY launch_date ASC LIMIT 3;

### 7
DIFFICULTY: join
QUESTION: List every show along with the name of the person who hosts it.
SQL: SELECT s.title, h.full_name FROM shows s JOIN hosts h ON h.host_id = s.host_id ORDER BY s.title;

### 8
DIFFICULTY: join
QUESTION: Which episodes belong to true crime shows, and what are those shows called?
SQL: SELECT e.title AS episode_title, s.title AS show_title FROM episodes e JOIN shows s ON s.show_id = e.show_id WHERE s.genre = 'True Crime' ORDER BY e.publish_date;

### 9
DIFFICULTY: join
QUESTION: Which companies have bought advertising on Tech Tonic?
SQL: SELECT sp.company_name, d.start_date, d.status FROM ad_deals d JOIN sponsors sp ON sp.sponsor_id = d.sponsor_id JOIN shows s ON s.show_id = d.show_id WHERE s.title = 'Tech Tonic' ORDER BY d.start_date;

### 10
DIFFICULTY: join
QUESTION: Which hosts on our roster don't currently host any show?
SQL: SELECT h.full_name, h.country FROM hosts h LEFT JOIN shows s ON s.host_id = h.host_id WHERE s.show_id IS NULL;

### 11
DIFFICULTY: join
QUESTION: Which shows have never published an episode?
SQL: SELECT s.title, s.launch_date FROM shows s LEFT JOIN episodes e ON e.show_id = s.show_id WHERE e.episode_id IS NULL;

### 12
DIFFICULTY: join
QUESTION: For episodes with a guest that came out in February or March 2023, who was the guest, which show was it on, and who hosts that show?
SQL: SELECT e.guest_name, e.title AS episode_title, s.title AS show_title, h.full_name AS host_name FROM episodes e JOIN shows s ON s.show_id = e.show_id JOIN hosts h ON h.host_id = s.host_id WHERE e.guest_name IS NOT NULL AND e.publish_date BETWEEN '2023-02-01' AND '2023-03-31' ORDER BY e.publish_date;

### 13
DIFFICULTY: aggregate
QUESTION: How many shows do we have in each genre, starting with the biggest genres?
SQL: SELECT genre, COUNT(*) AS show_count FROM shows GROUP BY genre ORDER BY show_count DESC, genre;

### 14
DIFFICULTY: aggregate
QUESTION: What are the total downloads for each show, from highest to lowest?
SQL: SELECT s.title, SUM(e.downloads) AS total_downloads FROM shows s JOIN episodes e ON e.show_id = s.show_id GROUP BY s.show_id, s.title ORDER BY total_downloads DESC;

### 15
DIFFICULTY: aggregate
QUESTION: What is the average episode length for each language we publish in?
SQL: SELECT s.language, ROUND(AVG(e.duration_minutes), 1) AS avg_minutes FROM episodes e JOIN shows s ON s.show_id = e.show_id GROUP BY s.language ORDER BY avg_minutes DESC;

### 16
DIFFICULTY: aggregate
QUESTION: How much advertising revenue is booked per sponsor industry, leaving out cancelled deals?
SQL: SELECT sp.industry, SUM(d.episodes_booked * d.price_per_episode) AS total_value FROM ad_deals d JOIN sponsors sp ON sp.sponsor_id = d.sponsor_id WHERE d.status <> 'cancelled' GROUP BY sp.industry ORDER BY total_value DESC;

### 17
DIFFICULTY: hard
QUESTION: Which episodes got more downloads than the average episode?
SQL: SELECT title, downloads FROM episodes WHERE downloads > (SELECT AVG(downloads) FROM episodes) ORDER BY downloads DESC;

### 18
DIFFICULTY: hard
QUESTION: Which sponsors have signed more than one ad deal, and how many?
SQL: SELECT sp.company_name, COUNT(*) AS deal_count FROM sponsors sp JOIN ad_deals d ON d.sponsor_id = sp.sponsor_id GROUP BY sp.sponsor_id, sp.company_name HAVING COUNT(*) > 1 ORDER BY deal_count DESC, sp.company_name;

### 19
DIFFICULTY: hard
QUESTION: How many episodes are short (under 30 minutes), medium (30 to 50 minutes) or long (over 50 minutes)?
SQL: SELECT CASE WHEN duration_minutes < 30 THEN 'Short' WHEN duration_minutes <= 50 THEN 'Medium' ELSE 'Long' END AS length_category, COUNT(*) AS episode_count FROM episodes GROUP BY length_category ORDER BY episode_count DESC;

### 20
DIFFICULTY: hard
QUESTION: Among ad deals that have an end date, which five ran the longest, and how many days did each last?
SQL: SELECT sp.company_name, d.start_date, d.end_date, CAST(julianday(d.end_date) - julianday(d.start_date) AS INTEGER) AS days_long FROM ad_deals d JOIN sponsors sp ON sp.sponsor_id = d.sponsor_id WHERE d.end_date IS NOT NULL ORDER BY days_long DESC LIMIT 5;
