PRAGMA foreign_keys = ON;

CREATE TABLE wind_farms (
    farm_id         INTEGER PRIMARY KEY,
    farm_name       TEXT NOT NULL,
    state           TEXT NOT NULL,
    commissioned_date TEXT,
    capacity_mw     REAL,
    operator        TEXT
);

CREATE TABLE turbines (
    turbine_id      INTEGER PRIMARY KEY,
    farm_id         INTEGER NOT NULL,
    turbine_tag     TEXT NOT NULL,
    model           TEXT,
    manufacturer    TEXT,
    rated_kw        INTEGER,
    install_date    TEXT,
    status          TEXT NOT NULL,
    FOREIGN KEY (farm_id) REFERENCES wind_farms(farm_id)
);

CREATE TABLE technicians (
    tech_id             INTEGER PRIMARY KEY,
    full_name           TEXT NOT NULL,
    certification_level TEXT NOT NULL,
    hire_date           TEXT,
    hourly_rate         REAL,
    home_base           TEXT
);

CREATE TABLE work_orders (
    work_order_id   INTEGER PRIMARY KEY,
    turbine_id      INTEGER NOT NULL,
    tech_id         INTEGER,
    order_type      TEXT NOT NULL,
    priority        TEXT NOT NULL,
    opened_date     TEXT NOT NULL,
    completed_date  TEXT,
    labor_hours     REAL,
    parts_cost      REAL,
    description     TEXT,
    FOREIGN KEY (turbine_id) REFERENCES turbines(turbine_id),
    FOREIGN KEY (tech_id) REFERENCES technicians(tech_id)
);

### 1
DIFFICULTY: simple
QUESTION: Which turbines are currently offline?
SQL: SELECT turbine_id, turbine_tag, model, status FROM turbines WHERE status = 'Offline';

### 2
DIFFICULTY: simple
QUESTION: Who are our five best-paid technicians?
SQL: SELECT full_name, hourly_rate FROM technicians ORDER BY hourly_rate DESC LIMIT 5;

### 3
DIFFICULTY: simple
QUESTION: Which maintenance jobs are still open, oldest first?
SQL: SELECT work_order_id, order_type, priority, opened_date FROM work_orders WHERE completed_date IS NULL ORDER BY opened_date ASC;

### 4
DIFFICULTY: simple
QUESTION: Show the wind farms in Texas or Wyoming, biggest capacity first.
SQL: SELECT farm_name, state, capacity_mw FROM wind_farms WHERE state IN ('Texas', 'Wyoming') ORDER BY capacity_mw DESC;

### 5
DIFFICULTY: simple
QUESTION: What were the three jobs with the highest spend on parts?
SQL: SELECT work_order_id, description, parts_cost FROM work_orders WHERE parts_cost IS NOT NULL ORDER BY parts_cost DESC LIMIT 3;

### 6
DIFFICULTY: simple
QUESTION: Which technicians joined the company in 2019 or later, in order of when they started?
SQL: SELECT full_name, hire_date, certification_level FROM technicians WHERE hire_date >= '2019-01-01' ORDER BY hire_date ASC;

### 7
DIFFICULTY: join
QUESTION: List the high-priority and critical jobs along with the name of the farm where each one happened.
SQL: SELECT w.work_order_id, w.priority, w.description, f.farm_name FROM work_orders w JOIN turbines t ON t.turbine_id = w.turbine_id JOIN wind_farms f ON f.farm_id = t.farm_id WHERE w.priority IN ('High', 'Critical') ORDER BY w.opened_date;

### 8
DIFFICULTY: join
QUESTION: Which technicians have worked on emergency repairs?
SQL: SELECT DISTINCT te.full_name FROM work_orders w JOIN technicians te ON te.tech_id = w.tech_id WHERE w.order_type = 'Emergency';

### 9
DIFFICULTY: join
QUESTION: What work has been done on Vestas turbines, and which turbine model was involved?
SQL: SELECT w.work_order_id, t.turbine_tag, t.model, w.order_type, w.description FROM work_orders w JOIN turbines t ON t.turbine_id = w.turbine_id WHERE t.manufacturer = 'Vestas' ORDER BY w.work_order_id;

### 10
DIFFICULTY: join
QUESTION: What jobs have been logged against turbines at our Iowa farms?
SQL: SELECT f.farm_name, t.turbine_tag, w.order_type, w.opened_date FROM wind_farms f JOIN turbines t ON t.farm_id = f.farm_id JOIN work_orders w ON w.turbine_id = t.turbine_id WHERE f.state = 'Iowa' ORDER BY w.opened_date;

### 11
DIFFICULTY: join
QUESTION: Which technicians have never been assigned to a job?
SQL: SELECT te.full_name, te.home_base FROM technicians te LEFT JOIN work_orders w ON w.tech_id = te.tech_id WHERE w.work_order_id IS NULL;

### 12
DIFFICULTY: join
QUESTION: For every unfinished job, show the turbine, the farm, the priority and who is handling it, saying "Unassigned" if nobody is.
SQL: SELECT w.work_order_id, t.turbine_tag, f.farm_name, w.priority, COALESCE(te.full_name, 'Unassigned') AS technician FROM work_orders w JOIN turbines t ON t.turbine_id = w.turbine_id JOIN wind_farms f ON f.farm_id = t.farm_id LEFT JOIN technicians te ON te.tech_id = w.tech_id WHERE w.completed_date IS NULL ORDER BY w.work_order_id;

### 13
DIFFICULTY: aggregate
QUESTION: How many jobs do we have of each type?
SQL: SELECT order_type, COUNT(*) AS job_count FROM work_orders GROUP BY order_type ORDER BY job_count DESC;

### 14
DIFFICULTY: aggregate
QUESTION: How much have we spent on parts at each farm in total?
SQL: SELECT f.farm_name, SUM(w.parts_cost) AS total_parts_cost FROM wind_farms f JOIN turbines t ON t.farm_id = f.farm_id JOIN work_orders w ON w.turbine_id = t.turbine_id GROUP BY f.farm_id, f.farm_name ORDER BY total_parts_cost DESC;

### 15
DIFFICULTY: aggregate
QUESTION: On average, how many labor hours does a job take at each priority level?
SQL: SELECT priority, ROUND(AVG(labor_hours), 2) AS avg_labor_hours FROM work_orders WHERE labor_hours IS NOT NULL GROUP BY priority ORDER BY avg_labor_hours DESC;

### 16
DIFFICULTY: aggregate
QUESTION: How many technicians are there at each certification level, and what is their average hourly pay?
SQL: SELECT certification_level, COUNT(*) AS technician_count, ROUND(AVG(hourly_rate), 2) AS avg_hourly_rate FROM technicians GROUP BY certification_level ORDER BY certification_level;

### 17
DIFFICULTY: hard
QUESTION: Which farms have spent more than $10,000 on parts overall?
SQL: SELECT f.farm_name, SUM(w.parts_cost) AS total_parts_cost FROM wind_farms f JOIN turbines t ON t.farm_id = f.farm_id JOIN work_orders w ON w.turbine_id = t.turbine_id GROUP BY f.farm_id, f.farm_name HAVING SUM(w.parts_cost) > 10000 ORDER BY total_parts_cost DESC;

### 18
DIFFICULTY: hard
QUESTION: For finished corrective and emergency repairs, how many days did each take from opening to completion, and was it slow (a week or more) or normal?
SQL: SELECT work_order_id, order_type, opened_date, completed_date, CAST(julianday(completed_date) - julianday(opened_date) AS INTEGER) AS days_to_complete, CASE WHEN julianday(completed_date) - julianday(opened_date) >= 7 THEN 'Slow' ELSE 'Normal' END AS speed FROM work_orders WHERE order_type IN ('Corrective', 'Emergency') AND completed_date IS NOT NULL ORDER BY days_to_complete DESC;

### 19
DIFFICULTY: hard
QUESTION: Which turbines have had more than one job logged against them?
SQL: SELECT t.turbine_tag, t.model, COUNT(*) AS job_count FROM turbines t JOIN work_orders w ON w.turbine_id = t.turbine_id GROUP BY t.turbine_id, t.turbine_tag, t.model HAVING COUNT(*) > 1;

### 20
DIFFICULTY: hard
QUESTION: Which technicians earn more per hour than the average technician?
SQL: SELECT full_name, hourly_rate FROM technicians WHERE hourly_rate > (SELECT AVG(hourly_rate) FROM technicians) ORDER BY hourly_rate DESC;
