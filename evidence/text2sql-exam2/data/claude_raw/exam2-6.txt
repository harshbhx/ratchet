PRAGMA foreign_keys = ON;

CREATE TABLE suppliers (
    supplier_id     INTEGER PRIMARY KEY,
    supplier_name   TEXT NOT NULL,
    supplier_type   TEXT NOT NULL,      -- Municipal, Commercial, Residential, Industrial
    city            TEXT NOT NULL,
    contract_start  TEXT NOT NULL,      -- ISO date
    contact_email   TEXT
);

CREATE TABLE materials (
    material_id      INTEGER PRIMARY KEY,
    material_name    TEXT NOT NULL,
    category         TEXT NOT NULL,     -- Plastic, Metal, Paper, Glass
    price_per_tonne  REAL NOT NULL
);

CREATE TABLE deliveries (
    delivery_id        INTEGER PRIMARY KEY,
    supplier_id        INTEGER NOT NULL,
    material_id        INTEGER NOT NULL,
    delivery_date      TEXT NOT NULL,   -- ISO date
    weight_kg          INTEGER NOT NULL,
    contamination_pct  REAL,
    truck_plate        TEXT,
    FOREIGN KEY (supplier_id) REFERENCES suppliers(supplier_id),
    FOREIGN KEY (material_id) REFERENCES materials(material_id)
);

CREATE TABLE processing_batches (
    batch_id     INTEGER PRIMARY KEY,
    material_id  INTEGER NOT NULL,
    batch_date   TEXT NOT NULL,         -- ISO date
    input_kg     INTEGER NOT NULL,
    output_kg    INTEGER,
    line_name    TEXT NOT NULL,
    supervisor   TEXT NOT NULL,
    status       TEXT NOT NULL,         -- Completed, In Progress, Rejected
    FOREIGN KEY (material_id) REFERENCES materials(material_id)
);

CREATE TABLE sales (
    sale_id         INTEGER PRIMARY KEY,
    material_id     INTEGER NOT NULL,
    buyer_name      TEXT NOT NULL,
    sale_date       TEXT NOT NULL,      -- ISO date
    weight_kg       INTEGER NOT NULL,
    total_revenue   REAL NOT NULL,
    payment_status  TEXT,
    FOREIGN KEY (material_id) REFERENCES materials(material_id)
);

### 1
DIFFICULTY: simple
QUESTION: Which of our suppliers are based in Portland? Please list them alphabetically.
SQL: SELECT supplier_name, supplier_type, contract_start FROM suppliers WHERE city = 'Portland' ORDER BY supplier_name;

### 2
DIFFICULTY: simple
QUESTION: What are the five biggest loads we've received, by weight?
SQL: SELECT delivery_id, delivery_date, weight_kg FROM deliveries ORDER BY weight_kg DESC LIMIT 5;

### 3
DIFFICULTY: simple
QUESTION: Which incoming loads never had a contamination reading taken?
SQL: SELECT delivery_id, delivery_date, weight_kg FROM deliveries WHERE contamination_pct IS NULL ORDER BY delivery_date;

### 4
DIFFICULTY: simple
QUESTION: Which metal materials are worth more than $1,000 a tonne?
SQL: SELECT material_name, price_per_tonne FROM materials WHERE category = 'Metal' AND price_per_tonne > 1000 ORDER BY price_per_tonne DESC;

### 5
DIFFICULTY: simple
QUESTION: Which processing runs were rejected or are still underway?
SQL: SELECT batch_id, batch_date, input_kg, line_name, status FROM processing_batches WHERE status IN ('Rejected', 'In Progress') ORDER BY batch_date;

### 6
DIFFICULTY: simple
QUESTION: Which sales have not been confirmed as paid yet?
SQL: SELECT sale_id, buyer_name, sale_date, total_revenue, payment_status FROM sales WHERE payment_status IS NULL OR payment_status <> 'Paid' ORDER BY sale_date;

### 7
DIFFICULTY: join
QUESTION: For the loads that arrived in March 2024, who sent them and what material was it?
SQL: SELECT d.delivery_date, s.supplier_name, m.material_name, d.weight_kg FROM deliveries d JOIN suppliers s ON s.supplier_id = d.supplier_id JOIN materials m ON m.material_id = d.material_id WHERE d.delivery_date BETWEEN '2024-03-01' AND '2024-03-31' ORDER BY d.delivery_date;

### 8
DIFFICULTY: join
QUESTION: What have our industrial suppliers brought in, and how much of it?
SQL: SELECT s.supplier_name, m.material_name, d.delivery_date, d.weight_kg FROM deliveries d JOIN suppliers s ON s.supplier_id = d.supplier_id JOIN materials m ON m.material_id = d.material_id WHERE s.supplier_type = 'Industrial' ORDER BY d.delivery_date;

### 9
DIFFICULTY: join
QUESTION: What materials were processed on Line C, and in what category do they fall?
SQL: SELECT b.batch_id, b.batch_date, m.material_name, m.category, b.input_kg FROM processing_batches b JOIN materials m ON m.material_id = b.material_id WHERE b.line_name = 'Line C' ORDER BY b.batch_date;

### 10
DIFFICULTY: join
QUESTION: Show me every sale of plastic material with the buyer and the type of plastic.
SQL: SELECT sa.sale_date, sa.buyer_name, m.material_name, sa.weight_kg, sa.total_revenue FROM sales sa JOIN materials m ON m.material_id = sa.material_id WHERE m.category = 'Plastic' ORDER BY sa.sale_date;

### 11
DIFFICULTY: join
QUESTION: Which suppliers have sent us glass?
SQL: SELECT DISTINCT s.supplier_name, s.city FROM deliveries d JOIN suppliers s ON s.supplier_id = d.supplier_id JOIN materials m ON m.material_id = d.material_id WHERE m.category = 'Glass' ORDER BY s.supplier_name;

### 12
DIFFICULTY: join
QUESTION: Which loads came in with more than 5% contamination, who delivered them, and which truck was it?
SQL: SELECT d.delivery_date, s.supplier_name, d.contamination_pct, d.truck_plate FROM deliveries d JOIN suppliers s ON s.supplier_id = d.supplier_id WHERE d.contamination_pct > 5 ORDER BY d.contamination_pct DESC;

### 13
DIFFICULTY: aggregate
QUESTION: How many kilograms have we received in total for each type of material category?
SQL: SELECT m.category, SUM(d.weight_kg) AS total_kg FROM deliveries d JOIN materials m ON m.material_id = d.material_id GROUP BY m.category ORDER BY total_kg DESC;

### 14
DIFFICULTY: aggregate
QUESTION: How many deliveries have we had from each kind of supplier?
SQL: SELECT s.supplier_type, COUNT(*) AS delivery_count FROM deliveries d JOIN suppliers s ON s.supplier_id = d.supplier_id GROUP BY s.supplier_type ORDER BY delivery_count DESC, s.supplier_type;

### 15
DIFFICULTY: aggregate
QUESTION: For each processing line, how many runs were there and how much material went in altogether?
SQL: SELECT line_name, COUNT(*) AS batch_count, SUM(input_kg) AS total_input_kg FROM processing_batches GROUP BY line_name ORDER BY total_input_kg DESC;

### 16
DIFFICULTY: aggregate
QUESTION: How much money has each buyer brought in, from highest to lowest?
SQL: SELECT buyer_name, SUM(total_revenue) AS revenue FROM sales GROUP BY buyer_name ORDER BY revenue DESC;

### 17
DIFFICULTY: hard
QUESTION: Which suppliers have an average contamination level higher than the average across all measured loads?
SQL: SELECT s.supplier_name, ROUND(AVG(d.contamination_pct), 2) AS avg_contamination FROM deliveries d JOIN suppliers s ON s.supplier_id = d.supplier_id GROUP BY s.supplier_id, s.supplier_name HAVING AVG(d.contamination_pct) > (SELECT AVG(contamination_pct) FROM deliveries) ORDER BY avg_contamination DESC;

### 18
DIFFICULTY: hard
QUESTION: For finished runs with a recorded output, what was the recovery rate, and how would you rate it as high (93% or more), medium (85% to under 93%) or low?
SQL: SELECT b.batch_id, m.material_name, ROUND(100.0 * b.output_kg / b.input_kg, 1) AS yield_pct, CASE WHEN 1.0 * b.output_kg / b.input_kg >= 0.93 THEN 'High' WHEN 1.0 * b.output_kg / b.input_kg >= 0.85 THEN 'Medium' ELSE 'Low' END AS yield_rating FROM processing_batches b JOIN materials m ON m.material_id = b.material_id WHERE b.status = 'Completed' AND b.output_kg IS NOT NULL ORDER BY yield_pct DESC;

### 19
DIFFICULTY: hard
QUESTION: For suppliers who have delivered more than once, how many days passed between their first and their most recent delivery?
SQL: SELECT s.supplier_name, MIN(d.delivery_date) AS first_delivery, MAX(d.delivery_date) AS last_delivery, CAST(julianday(MAX(d.delivery_date)) - julianday(MIN(d.delivery_date)) AS INTEGER) AS days_between FROM deliveries d JOIN suppliers s ON s.supplier_id = d.supplier_id GROUP BY s.supplier_id, s.supplier_name HAVING COUNT(*) > 1 ORDER BY days_between DESC;

### 20
DIFFICULTY: hard
QUESTION: Which materials have we received at the plant but never sold?
SQL: SELECT material_name, category FROM materials WHERE material_id IN (SELECT material_id FROM deliveries) AND material_id NOT IN (SELECT material_id FROM sales) ORDER BY material_name;
