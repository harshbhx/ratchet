CREATE TABLE cinemas (
    cinema_id      INTEGER PRIMARY KEY,
    name           TEXT NOT NULL,
    city           TEXT NOT NULL,
    opened_date    TEXT NOT NULL,
    screen_count   INTEGER NOT NULL,
    manager_name   TEXT
);

CREATE TABLE movies (
    movie_id       INTEGER PRIMARY KEY,
    title          TEXT NOT NULL,
    genre          TEXT NOT NULL,
    duration_min   INTEGER NOT NULL,
    rating         TEXT NOT NULL,
    release_date   TEXT NOT NULL,
    imdb_score     REAL
);

CREATE TABLE customers (
    customer_id    INTEGER PRIMARY KEY,
    full_name      TEXT NOT NULL,
    email          TEXT,
    join_date      TEXT NOT NULL,
    loyalty_tier   TEXT NOT NULL
);

CREATE TABLE screenings (
    screening_id   INTEGER PRIMARY KEY,
    cinema_id      INTEGER NOT NULL,
    movie_id       INTEGER NOT NULL,
    screen_number  INTEGER NOT NULL,
    start_time     TEXT NOT NULL,
    ticket_price   REAL NOT NULL,
    seats_sold     INTEGER NOT NULL,
    FOREIGN KEY (cinema_id) REFERENCES cinemas(cinema_id),
    FOREIGN KEY (movie_id) REFERENCES movies(movie_id)
);

CREATE TABLE bookings (
    booking_id      INTEGER PRIMARY KEY,
    screening_id    INTEGER NOT NULL,
    customer_id     INTEGER,
    tickets_count   INTEGER NOT NULL,
    total_paid      REAL NOT NULL,
    booking_date    TEXT NOT NULL,
    payment_method  TEXT NOT NULL,
    FOREIGN KEY (screening_id) REFERENCES screenings(screening_id),
    FOREIGN KEY (customer_id) REFERENCES customers(customer_id)
);

### 1
DIFFICULTY: simple
QUESTION: Which movies are suitable for all ages (rated G), and how long are they?
SQL: SELECT title, duration_min FROM movies WHERE rating = 'G';

### 2
DIFFICULTY: simple
QUESTION: What are the five longest movies we show?
SQL: SELECT title, duration_min FROM movies ORDER BY duration_min DESC LIMIT 5;

### 3
DIFFICULTY: simple
QUESTION: Which of our cinemas currently have no manager on record?
SQL: SELECT name, city FROM cinemas WHERE manager_name IS NULL;

### 4
DIFFICULTY: simple
QUESTION: Show me our Gold and Platinum members in alphabetical order.
SQL: SELECT full_name, loyalty_tier FROM customers WHERE loyalty_tier IN ('Gold', 'Platinum') ORDER BY full_name;

### 5
DIFFICULTY: simple
QUESTION: What are the three highest ticket prices charged for a showing?
SQL: SELECT screening_id, start_time, ticket_price FROM screenings ORDER BY ticket_price DESC LIMIT 3;

### 6
DIFFICULTY: simple
QUESTION: Which films opened between April and June 2024, in order of release?
SQL: SELECT title, release_date FROM movies WHERE release_date BETWEEN '2024-04-01' AND '2024-06-30' ORDER BY release_date;

### 7
DIFFICULTY: join
QUESTION: Which movies were shown in July 2024 or later, and at which cinemas?
SQL: SELECT m.title, c.name AS cinema, s.start_time FROM screenings s JOIN movies m ON m.movie_id = s.movie_id JOIN cinemas c ON c.cinema_id = s.cinema_id WHERE s.start_time >= '2024-07-01' ORDER BY s.start_time;

### 8
DIFFICULTY: join
QUESTION: Who has booked tickets to see Galaxy Drifters, and how many seats did each booking cover?
SQL: SELECT cu.full_name, b.tickets_count, b.booking_date FROM bookings b JOIN customers cu ON cu.customer_id = b.customer_id JOIN screenings s ON s.screening_id = b.screening_id JOIN movies m ON m.movie_id = s.movie_id WHERE m.title = 'Galaxy Drifters' ORDER BY b.booking_date;

### 9
DIFFICULTY: join
QUESTION: What was shown at our Chicago locations, and what did a ticket cost?
SQL: SELECT c.name AS cinema, m.title, s.start_time, s.ticket_price FROM cinemas c JOIN screenings s ON s.cinema_id = c.cinema_id JOIN movies m ON m.movie_id = s.movie_id WHERE c.city = 'Chicago' ORDER BY s.start_time;

### 10
DIFFICULTY: join
QUESTION: For bookings paid through the mobile app, who paid and which film was it for?
SQL: SELECT cu.full_name, m.title, b.total_paid FROM bookings b JOIN customers cu ON cu.customer_id = b.customer_id JOIN screenings s ON s.screening_id = b.screening_id JOIN movies m ON m.movie_id = s.movie_id WHERE b.payment_method = 'mobile' ORDER BY b.total_paid DESC;

### 11
DIFFICULTY: join
QUESTION: Which cinemas have screened a sci-fi film?
SQL: SELECT DISTINCT c.name, c.city FROM cinemas c JOIN screenings s ON s.cinema_id = c.cinema_id JOIN movies m ON m.movie_id = s.movie_id WHERE m.genre = 'Sci-Fi' ORDER BY c.name;

### 12
DIFFICULTY: join
QUESTION: List every booking with the buyer's name (or "Guest" if unknown), the film and the cinema.
SQL: SELECT b.booking_id, COALESCE(cu.full_name, 'Guest') AS buyer, m.title, c.name AS cinema, b.total_paid FROM bookings b LEFT JOIN customers cu ON cu.customer_id = b.customer_id JOIN screenings s ON s.screening_id = b.screening_id JOIN movies m ON m.movie_id = s.movie_id JOIN cinemas c ON c.cinema_id = s.cinema_id ORDER BY b.booking_id;

### 13
DIFFICULTY: aggregate
QUESTION: How many movies do we have in each genre?
SQL: SELECT genre, COUNT(*) AS movie_count FROM movies GROUP BY genre ORDER BY movie_count DESC, genre;

### 14
DIFFICULTY: aggregate
QUESTION: How much money came in through each payment method, and how many bookings were there?
SQL: SELECT payment_method, COUNT(*) AS bookings, SUM(total_paid) AS revenue FROM bookings GROUP BY payment_method ORDER BY revenue DESC;

### 15
DIFFICULTY: aggregate
QUESTION: What is the average ticket price in each city?
SQL: SELECT c.city, ROUND(AVG(s.ticket_price), 2) AS avg_price FROM cinemas c JOIN screenings s ON s.cinema_id = c.cinema_id GROUP BY c.city ORDER BY avg_price DESC;

### 16
DIFFICULTY: aggregate
QUESTION: Which five movies have sold the most seats overall?
SQL: SELECT m.title, SUM(s.seats_sold) AS total_seats FROM movies m JOIN screenings s ON s.movie_id = m.movie_id GROUP BY m.movie_id, m.title ORDER BY total_seats DESC LIMIT 5;

### 17
DIFFICULTY: hard
QUESTION: Which movies have a critic score higher than the average across all scored movies?
SQL: SELECT title, imdb_score FROM movies WHERE imdb_score > (SELECT AVG(imdb_score) FROM movies) ORDER BY imdb_score DESC;

### 18
DIFFICULTY: hard
QUESTION: Which customers have spent more than $50 in total on bookings, and how much?
SQL: SELECT cu.full_name, SUM(b.total_paid) AS total_spent FROM customers cu JOIN bookings b ON b.customer_id = cu.customer_id GROUP BY cu.customer_id, cu.full_name HAVING SUM(b.total_paid) > 50 ORDER BY total_spent DESC;

### 19
DIFFICULTY: hard
QUESTION: How do daytime showings (starting before 5 PM) compare with evening ones in number of showings and seats sold?
SQL: SELECT CASE WHEN CAST(strftime('%H', start_time) AS INTEGER) < 17 THEN 'Matinee' ELSE 'Evening' END AS slot, COUNT(*) AS showings, SUM(seats_sold) AS total_seats FROM screenings GROUP BY slot;

### 20
DIFFICULTY: hard
QUESTION: Which five cinemas have been open the longest, and how many full years had they been open as of the end of 2024?
SQL: SELECT name, city, opened_date, CAST((julianday('2024-12-31') - julianday(opened_date)) / 365.25 AS INTEGER) AS years_open FROM cinemas ORDER BY opened_date ASC LIMIT 5;
