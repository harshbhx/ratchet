PRAGMA foreign_keys = ON;

CREATE TABLE producers (
    producer_id       INTEGER PRIMARY KEY,
    name              TEXT NOT NULL,
    country           TEXT NOT NULL,
    region            TEXT,
    founded_year      INTEGER,
    organic_certified INTEGER  -- 1 = yes, 0 = no, NULL = unknown
);

CREATE TABLE wines (
    wine_id      INTEGER PRIMARY KEY,
    producer_id  INTEGER NOT NULL,
    wine_name    TEXT NOT NULL,
    varietal     TEXT,
    vintage      INTEGER,        -- NULL for non-vintage
    color        TEXT NOT NULL,  -- Red, White, Rose, Sparkling
    bottle_price REAL NOT NULL,  -- wholesale cost per bottle
    alcohol_pct  REAL,
    FOREIGN KEY (producer_id) REFERENCES producers(producer_id)
);

CREATE TABLE customers (
    customer_id    INTEGER PRIMARY KEY,
    business_name  TEXT NOT NULL,
    customer_type  TEXT NOT NULL,  -- Restaurant, Retail Shop, Bar, Hotel
    city           TEXT,
    state          TEXT,
    account_opened TEXT,           -- ISO date
    credit_limit   REAL
);

CREATE TABLE orders (
    order_id    INTEGER PRIMARY KEY,
    customer_id INTEGER NOT NULL,
    order_date  TEXT NOT NULL,   -- ISO date
    ship_date   TEXT,            -- ISO date, NULL if not shipped
    status      TEXT NOT NULL,   -- Delivered, Shipped, Pending, Cancelled
    FOREIGN KEY (customer_id) REFERENCES customers(customer_id)
);

CREATE TABLE order_items (
    item_id      INTEGER PRIMARY KEY,
    order_id     INTEGER NOT NULL,
    wine_id      INTEGER NOT NULL,
    quantity     INTEGER NOT NULL,  -- bottles
    unit_price   REAL NOT NULL,     -- price charged per bottle
    discount_pct REAL,              -- NULL = no discount
    FOREIGN KEY (order_id) REFERENCES orders(order_id),
    FOREIGN KEY (wine_id) REFERENCES wines(wine_id)
);

### 1
DIFFICULTY: simple
QUESTION: Which wines cost us more than $25 a bottle, starting with the priciest?
SQL: SELECT wine_name, varietal, bottle_price FROM wines WHERE bottle_price > 25 ORDER BY bottle_price DESC;

### 2
DIFFICULTY: simple
QUESTION: Who are our three newest customers?
SQL: SELECT business_name, city, account_opened FROM customers ORDER BY account_opened DESC LIMIT 3;

### 3
DIFFICULTY: simple
QUESTION: Which orders are still pending and haven't gone out the door yet?
SQL: SELECT order_id, customer_id, order_date FROM orders WHERE status = 'Pending' AND ship_date IS NULL ORDER BY order_date;

### 4
DIFFICULTY: simple
QUESTION: Which customers don't have a credit limit set up?
SQL: SELECT business_name, customer_type, city FROM customers WHERE credit_limit IS NULL;

### 5
DIFFICULTY: simple
QUESTION: Which American wineries hold an organic certification?
SQL: SELECT name, region, founded_year FROM producers WHERE country = 'USA' AND organic_certified = 1;

### 6
DIFFICULTY: simple
QUESTION: What white wines from the 2022 vintage do we carry?
SQL: SELECT wine_name, varietal, bottle_price FROM wines WHERE color = 'White' AND vintage = 2022 ORDER BY wine_name;

### 7
DIFFICULTY: join
QUESTION: Which wines do we get from French producers, and who makes them?
SQL: SELECT p.name AS producer, w.wine_name FROM producers p JOIN wines w ON w.producer_id = p.producer_id WHERE p.country = 'France' ORDER BY p.name;

### 8
DIFFICULTY: join
QUESTION: Which businesses placed orders from July 2023 onward, and on what dates?
SQL: SELECT o.order_id, c.business_name, o.order_date, o.status FROM orders o JOIN customers c ON c.customer_id = o.customer_id WHERE o.order_date >= '2023-07-01' ORDER BY o.order_date;

### 9
DIFFICULTY: join
QUESTION: What wines has Harbor Wine & Spirits bought from us?
SQL: SELECT DISTINCT w.wine_name, w.varietal FROM customers c JOIN orders o ON o.customer_id = c.customer_id JOIN order_items oi ON oi.order_id = o.order_id JOIN wines w ON w.wine_id = oi.wine_id WHERE c.business_name = 'Harbor Wine & Spirits' ORDER BY w.wine_name;

### 10
DIFFICULTY: join
QUESTION: Which restaurants have placed orders, and when did they order?
SQL: SELECT c.business_name, o.order_id, o.order_date FROM customers c JOIN orders o ON o.customer_id = c.customer_id WHERE c.customer_type = 'Restaurant' ORDER BY c.business_name, o.order_date;

### 11
DIFFICULTY: join
QUESTION: How many bottles of Italian wine have been ordered on each order line, and from which producer?
SQL: SELECT w.wine_name, p.name AS producer, oi.order_id, oi.quantity FROM order_items oi JOIN wines w ON w.wine_id = oi.wine_id JOIN producers p ON p.producer_id = w.producer_id WHERE p.country = 'Italy' ORDER BY oi.order_id;

### 12
DIFFICULTY: join
QUESTION: Which customers have never placed an order?
SQL: SELECT c.business_name, c.customer_type, c.city FROM customers c LEFT JOIN orders o ON o.customer_id = c.customer_id WHERE o.order_id IS NULL;

### 13
DIFFICULTY: aggregate
QUESTION: How many customers do we have of each business type?
SQL: SELECT customer_type, COUNT(*) AS customer_count FROM customers GROUP BY customer_type ORDER BY customer_count DESC;

### 14
DIFFICULTY: aggregate
QUESTION: What is the average bottle cost for each wine color?
SQL: SELECT color, ROUND(AVG(bottle_price), 2) AS avg_price FROM wines GROUP BY color ORDER BY avg_price DESC;

### 15
DIFFICULTY: aggregate
QUESTION: How many bottles are tied up in orders under each order status?
SQL: SELECT o.status, SUM(oi.quantity) AS total_bottles FROM orders o JOIN order_items oi ON oi.order_id = o.order_id GROUP BY o.status ORDER BY total_bottles DESC;

### 16
DIFFICULTY: aggregate
QUESTION: How many different wines do we stock from each producing country?
SQL: SELECT p.country, COUNT(*) AS wine_count FROM wines w JOIN producers p ON p.producer_id = w.producer_id GROUP BY p.country ORDER BY wine_count DESC, p.country;

### 17
DIFFICULTY: hard
QUESTION: Which customers have spent more than $1,000 on non-cancelled orders once discounts are taken off, and how much was it?
SQL: SELECT c.business_name, ROUND(SUM(oi.quantity * oi.unit_price * (1 - COALESCE(oi.discount_pct, 0) / 100.0)), 2) AS total_spent FROM customers c JOIN orders o ON o.customer_id = c.customer_id JOIN order_items oi ON oi.order_id = o.order_id WHERE o.status <> 'Cancelled' GROUP BY c.customer_id, c.business_name HAVING total_spent > 1000 ORDER BY total_spent DESC;

### 18
DIFFICULTY: hard
QUESTION: Which wines in our catalog have never appeared on any order?
SQL: SELECT wine_name, color, bottle_price FROM wines WHERE wine_id NOT IN (SELECT wine_id FROM order_items) ORDER BY wine_name;

### 19
DIFFICULTY: hard
QUESTION: What are the five slowest orders to ship, measured in days between ordering and shipping?
SQL: SELECT order_id, order_date, ship_date, CAST(julianday(ship_date) - julianday(order_date) AS INTEGER) AS days_to_ship FROM orders WHERE ship_date IS NOT NULL ORDER BY days_to_ship DESC, order_id LIMIT 5;

### 20
DIFFICULTY: hard
QUESTION: How many customers fall into each credit tier: high (30,000 or more), medium (10,000 to under 30,000), low (under 10,000), or none set?
SQL: SELECT CASE WHEN credit_limit IS NULL THEN 'No limit set' WHEN credit_limit >= 30000 THEN 'High' WHEN credit_limit >= 10000 THEN 'Medium' ELSE 'Low' END AS credit_tier, COUNT(*) AS customer_count FROM customers GROUP BY credit_tier ORDER BY customer_count DESC;
