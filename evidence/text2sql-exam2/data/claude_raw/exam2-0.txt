PRAGMA foreign_keys = ON;

CREATE TABLE galleries (
    gallery_id   INTEGER PRIMARY KEY,
    name         TEXT NOT NULL,
    floor        INTEGER NOT NULL,
    area_sqm     REAL
);

CREATE TABLE artists (
    artist_id    INTEGER PRIMARY KEY,
    full_name    TEXT NOT NULL,
    nationality  TEXT,
    birth_year   INTEGER,
    death_year   INTEGER
);

CREATE TABLE artworks (
    artwork_id        INTEGER PRIMARY KEY,
    title             TEXT NOT NULL,
    artist_id         INTEGER NOT NULL,
    creation_year     INTEGER,
    medium            TEXT,
    acquisition_date  TEXT,
    acquisition_cost  REAL,
    condition_rating  INTEGER,
    gallery_id        INTEGER,
    FOREIGN KEY (artist_id) REFERENCES artists(artist_id),
    FOREIGN KEY (gallery_id) REFERENCES galleries(gallery_id)
);

CREATE TABLE exhibitions (
    exhibition_id  INTEGER PRIMARY KEY,
    title          TEXT NOT NULL,
    start_date     TEXT NOT NULL,
    end_date       TEXT,
    gallery_id     INTEGER NOT NULL,
    ticket_price   REAL,
    FOREIGN KEY (gallery_id) REFERENCES galleries(gallery_id)
);

CREATE TABLE exhibition_artworks (
    exhibition_id  INTEGER NOT NULL,
    artwork_id     INTEGER NOT NULL,
    insured_value  REAL,
    PRIMARY KEY (exhibition_id, artwork_id),
    FOREIGN KEY (exhibition_id) REFERENCES exhibitions(exhibition_id),
    FOREIGN KEY (artwork_id) REFERENCES artworks(artwork_id)
);

### 1
DIFFICULTY: simple
QUESTION: Which oil paintings on canvas came into the collection before 1970, and when were they acquired?
SQL: SELECT title, acquisition_date FROM artworks WHERE medium = 'Oil on canvas' AND acquisition_date < '1970-01-01' ORDER BY acquisition_date;

### 2
DIFFICULTY: simple
QUESTION: What are the five most expensive purchases we have made?
SQL: SELECT title, acquisition_cost FROM artworks WHERE acquisition_cost IS NOT NULL ORDER BY acquisition_cost DESC LIMIT 5;

### 3
DIFFICULTY: simple
QUESTION: List the rooms on the second floor from largest to smallest, with their sizes.
SQL: SELECT name, area_sqm FROM galleries WHERE floor = 2 ORDER BY area_sqm DESC;

### 4
DIFFICULTY: simple
QUESTION: Which exhibitions have no ticket price set?
SQL: SELECT title, start_date, end_date FROM exhibitions WHERE ticket_price IS NULL;

### 5
DIFFICULTY: simple
QUESTION: Which known living artists, with a recorded nationality, are represented in our records?
SQL: SELECT full_name, nationality, birth_year FROM artists WHERE death_year IS NULL AND nationality IS NOT NULL ORDER BY full_name;

### 6
DIFFICULTY: simple
QUESTION: Which pieces are in poor condition (rated 2 or lower), worst first?
SQL: SELECT title, condition_rating FROM artworks WHERE condition_rating <= 2 ORDER BY condition_rating, title;

### 7
DIFFICULTY: join
QUESTION: Which pieces in the collection were made by American artists, and who made them?
SQL: SELECT aw.title, ar.full_name FROM artworks aw JOIN artists ar ON ar.artist_id = aw.artist_id WHERE ar.nationality = 'American' ORDER BY aw.title;

### 8
DIFFICULTY: join
QUESTION: What is on display in rooms on the top floor (floor 3), and in which room is each piece?
SQL: SELECT aw.title, g.name AS room FROM artworks aw JOIN galleries g ON g.gallery_id = aw.gallery_id WHERE g.floor = 3 ORDER BY g.name;

### 9
DIFFICULTY: join
QUESTION: Which pieces are not currently assigned to any room, and who are the artists?
SQL: SELECT aw.title, ar.full_name FROM artworks aw JOIN artists ar ON ar.artist_id = aw.artist_id WHERE aw.gallery_id IS NULL;

### 10
DIFFICULTY: join
QUESTION: Which artworks were shown in the Modern Masters exhibition, and what was each insured for?
SQL: SELECT aw.title, ea.insured_value FROM exhibition_artworks ea JOIN exhibitions e ON e.exhibition_id = ea.exhibition_id JOIN artworks aw ON aw.artwork_id = ea.artwork_id WHERE e.title = 'Modern Masters';

### 11
DIFFICULTY: join
QUESTION: For exhibitions charging more than 15 for entry, which room hosted each one?
SQL: SELECT e.title, g.name AS room, e.ticket_price FROM exhibitions e JOIN galleries g ON g.gallery_id = e.gallery_id WHERE e.ticket_price > 15 ORDER BY e.ticket_price DESC;

### 12
DIFFICULTY: join
QUESTION: Which exhibitions have no artworks recorded against them yet?
SQL: SELECT e.title, e.start_date FROM exhibitions e LEFT JOIN exhibition_artworks ea ON ea.exhibition_id = e.exhibition_id WHERE ea.artwork_id IS NULL;

### 13
DIFFICULTY: aggregate
QUESTION: How many pieces do we hold in each type of medium?
SQL: SELECT medium, COUNT(*) AS piece_count FROM artworks GROUP BY medium ORDER BY piece_count DESC, medium;

### 14
DIFFICULTY: aggregate
QUESTION: How much have we spent acquiring works from artists of each nationality?
SQL: SELECT ar.nationality, SUM(aw.acquisition_cost) AS total_spent FROM artworks aw JOIN artists ar ON ar.artist_id = aw.artist_id GROUP BY ar.nationality ORDER BY total_spent DESC;

### 15
DIFFICULTY: aggregate
QUESTION: What is the average ticket price of exhibitions on each floor of the building?
SQL: SELECT g.floor, AVG(e.ticket_price) AS avg_price FROM exhibitions e JOIN galleries g ON g.gallery_id = e.gallery_id GROUP BY g.floor ORDER BY g.floor;

### 16
DIFFICULTY: aggregate
QUESTION: How many exhibitions opened in each year?
SQL: SELECT strftime('%Y', start_date) AS year, COUNT(*) AS exhibitions_opened FROM exhibitions GROUP BY year ORDER BY year;

### 17
DIFFICULTY: hard
QUESTION: Which pieces cost more to acquire than the average purchase price across the collection?
SQL: SELECT title, acquisition_cost FROM artworks WHERE acquisition_cost > (SELECT AVG(acquisition_cost) FROM artworks) ORDER BY acquisition_cost DESC;

### 18
DIFFICULTY: hard
QUESTION: Which artists have more than one piece in our collection, and how many do we have?
SQL: SELECT ar.full_name, COUNT(*) AS works_held FROM artists ar JOIN artworks aw ON aw.artist_id = ar.artist_id GROUP BY ar.artist_id, ar.full_name HAVING COUNT(*) > 1 ORDER BY works_held DESC, ar.full_name;

### 19
DIFFICULTY: hard
QUESTION: How many pieces come from each broad period: before 1800, the 1800s, the 1900s, and the 2000s onward?
SQL: SELECT CASE WHEN creation_year < 1800 THEN 'Before 1800' WHEN creation_year < 1900 THEN '19th century' WHEN creation_year < 2000 THEN '20th century' ELSE '21st century' END AS period, COUNT(*) AS piece_count FROM artworks WHERE creation_year IS NOT NULL GROUP BY period ORDER BY MIN(creation_year);

### 20
DIFFICULTY: hard
QUESTION: Among exhibitions that have already ended, which five ran the longest, and how many days did each last?
SQL: SELECT title, CAST(julianday(end_date) - julianday(start_date) AS INTEGER) AS days_open FROM exhibitions WHERE end_date IS NOT NULL ORDER BY days_open DESC LIMIT 5;
